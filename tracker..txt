/**
 * OOP: Structural Record Entry Model
 */
class MissingRecord {
  constructor(id, entityType, name, location, datetime, extraDetail, revisionNum = 1) {
    this.id = id.trim().toUpperCase();
    this.entityType = entityType;
    this.name = name.trim();
    this.location = location.trim();
    this.datetime = datetime.replace('T', ' ');
    this.extraDetail = extraDetail.trim() || 'N/A';
    this.revisionNum = revisionNum;
  }
}

/**
 * DSA & OOP: Historical Map Engine (No seed data, starts clean!)
 */
class HistoricalRecordDatabase {
  constructor() {
    this.storageKey = 'missing_tracker_clean_v2';
    this.recordsMap = this.loadFromStorage();
  }

  loadFromStorage() {
    const data = localStorage.getItem(this.storageKey);
    return data ? JSON.parse(data) : {};
  }

  persist() {
    localStorage.setItem(this.storageKey, JSON.stringify(this.recordsMap));
  }

  save(record) {
    if (!this.recordsMap[record.id]) {
      this.recordsMap[record.id] = [];
    }
    
    record.revisionNum = this.recordsMap[record.id].length + 1;
    this.recordsMap[record.id].push(record);
    this.persist();
  }

  search(query) {
    const lowerQuery = query.toLowerCase().trim();
    const allHistoricalRows = Object.values(this.recordsMap).flat();
    
    // Sort array globally by date descending (Newest updates appear first)
    allHistoricalRows.sort((a, b) => new Date(b.datetime) - new Date(a.datetime));

    if (!lowerQuery) return allHistoricalRows;

    return allHistoricalRows.filter(item => 
      item.id.toLowerCase().includes(lowerQuery) ||
      item.name.toLowerCase().includes(lowerQuery) ||
      item.entityType.toLowerCase().includes(lowerQuery) ||
      item.location.toLowerCase().includes(lowerQuery) ||
      item.extraDetail.toLowerCase().includes(lowerQuery)
    );
  }
}

const DB = new HistoricalRecordDatabase();

function renderTable(recordsList) {
  const tbody = document.getElementById('recordTableBody');
  tbody.innerHTML = '';

  if (recordsList.length === 0) {
    tbody.innerHTML = `<tr><td colspan="6" class="no-results">No log histories found. Add your first record above!</td></tr>`;
    return;
  }

  recordsList.forEach(item => {
    const row = document.createElement('tr');
    const badgeClass = item.revisionNum === 1 ? 'initial-badge' : 'update-badge';
    const badgeText = item.revisionNum === 1 ? 'Initial' : `Update v${item.revisionNum}`;

    row.innerHTML = `
      <td>
        <strong>${item.id}</strong> 
        <span class="revision-badge ${badgeClass}">${badgeText}</span>
      </td>
      <td>${item.entityType}</td>
      <td>${item.name}</td>
      <td><strong>${item.location}</strong></td>
      <td>${item.datetime}</td>
      <td><em>${item.extraDetail}</em></td>
    `;
    tbody.appendChild(row);
  });
}

// Controller Listeners
document.getElementById('recordForm').addEventListener('submit', function(e) {
  e.preventDefault();

  const entry = new MissingRecord(
    document.getElementById('recordId').value,
    document.getElementById('entityType').value,
    document.getElementById('name').value,
    document.getElementById('location').value,
    document.getElementById('datetime').value,
    document.getElementById('extraDetail').value
  );

  DB.save(entry);
  renderTable(DB.search(document.getElementById('searchEngine').value));
  this.reset();
});

document.getElementById('searchEngine').addEventListener('input', function(e) {
  renderTable(DB.search(e.target.value));
});

// Boot Initializer 
renderTable(DB.search(''));
